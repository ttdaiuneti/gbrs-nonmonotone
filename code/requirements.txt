# Reproducibility environment for
# "Non-Monotonicity in Granular-Ball Attribute Reduction"
#
# The published numbers (all CSVs under supplementary_data/ and results/)

numpy>=1.26,<3
scipy>=1.11
pandas>=2.0
matplotlib>=3.7
